    48→import { TelecomRechargePage } from "@/components/TelecomRechargePage";

    69→import { useLanguage } from "@/context/LanguageContext";
    70→
    71→const APP_VERSION = "2.1.0";
    72→
    73→export default function AppleNetApp() {
    74→  const { t, isRTL, lang } = useLanguage();
    75→
    76→  const NAV_TABS = [
    77→    { id: "home", icon: Home, label: t("nav.home") },
    78→    { id: "cards", icon: Wifi, label: t("nav.cards") },
    79→    { id: "starlink", icon: Satellite, label: "Starlink" },
    80→    { id: "chat", icon: MessageCircle, label: t("nav.chat") },
    81→    { id: "credit", icon: Wallet, label: t("nav.credit") },
    82→  ];
    83→
    84→  const [user, setUser] = useState<FirebaseUser | null>(null);
    85→  const [userRole, setUserRole] = useState<string>("user");
    86→  const [activeTab, setActiveTab] = useState("home");
    87→  const [showAuth, setShowAuth] = useState(false);
    88→  const [authMode, setAuthMode] = useState<"login" | "register">("login");
    89→  const [showAdmin, setShowAdmin] = useState(false);
    90→  const [loading, setLoading] = useState(true);
    91→  const [showMenu, setShowMenu] = useState(false);
    92→  const [splashDone, setSplashDone] = useState(false);
    93→  const [splashExiting, setSplashExiting] = useState(false);
    94→  const [userName, setUserName] = useState<string>("");
    95→  const [showOnboarding, setShowOnboarding] = useState(false);
    96→  const [splashProgress, setSplashProgress] = useState(0);
    97→  const [showLocationModal, setShowLocationModal] = useState(false);
    98→  const [locationModalProvinceId, setLocationModalProvinceId] = useState("");
    99→  const [locationModalDistrict, setLocationModalDistrict] = useState("");
   100→  const [locationModalSaving, setLocationModalSaving] = useState(false);
   101→  const [locationModalDismissed, setLocationModalDismissed] = useState(false);
   102→  const [userPhotoURL, setUserPhotoURL] = useState<string>("");
   103→  const [pinLocked, setPinLocked] = useState(false);
   104→  const [biometricEnabled, setBiometricEnabled] = useState(false);
   105→  const [pinEnabled, setPinEnabled] = useState(false);
   106→
   107→  // Listen for PIN/biometric setting changes
   108→  useEffect(() => {
   109→    if (!user) return;
   110→    const pinUnsub = onValue(ref(db, `users/${user.uid}/pinEnabled`), (snap) => {
   111→      setPinEnabled(!!snap.val());
   112→    });
   113→    const bioUnsub = onValue(ref(db, `users/${user.uid}/biometricEnabled`), (snap) => {
   114→      setBiometricEnabled(!!snap.val());
   115→    });
   116→    return () => { pinUnsub(); bioUnsub(); };
   117→  }, [user]);
   118→
   119→  // Handle URL params for PWA shortcuts
   120→  useEffect(() => {
   121→    if (typeof window === "undefined") return;
   122→    const params = new URLSearchParams(window.location.search);
   123→    const tab = params.get("tab");
   124→    if (tab && (NAV_TABS.some(t => t.id === tab) || tab === "profile" || tab === "deposit" || tab === "purchased" || tab === "submit-network")) {
   125→      setActiveTab(tab);
   126→    }
   127→  }, []);
   128→
   129→  useEffect(() => {
   130→    const unsub = onAuthStateChanged(auth, async (u) => {
   131→      setUser(u);
   132→      if (u) {
   133→        try {
   134→          const roleSnap = await get(ref(db, `users/${u.uid}/role`));
   135→          const role = roleSnap.val() || "user";
   136→          setUserRole(role);
   137→          // Admin check removed from user app
   138→          const nameSnap = await get(ref(db, `users/${u.uid}/displayName`));
   139→          setUserName(nameSnap.val() || u.displayName || u.email?.split("@")[0] || "م");
   140→          // Check if user has provinceId, show location modal if not
   141→          const provinceSnap = await get(ref(db, `users/${u.uid}/provinceId`));
   142→          const hasProvince = provinceSnap.exists() && provinceSnap.val();
   143→          if (!hasProvince && !locationModalDismissed) {
   144→            setShowLocationModal(true);
   145→          }
   146→          // Network manager check removed from user app
   147→          // Check PIN and biometric settings
   148→          const pinSnap = await get(ref(db, `users/${u.uid}/pinEnabled`));
   149→          const bioSnap = await get(ref(db, `users/${u.uid}/biometricEnabled`));
   150→          const isPinEnabled = !!pinSnap.val();
   151→          const isBioEnabled = !!bioSnap.val();
   152→          setPinEnabled(isPinEnabled);
   153→          setBiometricEnabled(isBioEnabled);
   154→          if (isPinEnabled) setPinLocked(true);
   155→          // Initialize FCM
   156→          initFCMToken(u.uid).catch(() => {});
   157→        } catch {
   158→          setIsAdmin(false);
   159→          setUserRole("user");
   160→        }
   161→      } else {
   162→        setIsAdmin(false);
   163→        setUserRole("user");
   164→        setManagedNetwork("");
   165→        setUserName("");
   166→      }
   167→      setLoading(false);
   168→    });
   169→    return () => unsub();
   170→  }, []);
   171→
   172→  useEffect(() => {
   173→    if (!user) return;
   174→    const roleUnsub = onValue(ref(db, `users/${user.uid}/role`), (snap) => {
   175→      const role = snap.val() || "user";
   176→      setUserRole(role);
   177→      setIsAdmin(role === "admin");
   178→      if (role === "network_manager") {
   179→        get(ref(db, `users/${user.uid}/managedNetwork`)).then((netSnap) => {
   180→          setManagedNetwork(netSnap.val() || "");
   181→        });
   182→      } else {
   183→        setManagedNetwork("");
   184→      }
   185→    });
   186→    const nameUnsub = onValue(ref(db, `users/${user.uid}/displayName`), (snap) => {
   187→      if (snap.val()) setUserName(snap.val());
   188→    });
   189→    return () => { roleUnsub(); nameUnsub(); };
   190→  }, [user]);
   191→
   192→  // Listen for user photo URL
   193→  useEffect(() => {
   194→    if (!user) return;
   195→    const photoUnsub = onValue(ref(db, `users/${user.uid}/photoURL`), (snap) => {
   196→      if (snap.val()) setUserPhotoURL(snap.val());
   197→      else setUserPhotoURL("");
   198→    });
   199→    return () => photoUnsub();
   200→  }, [user]);
   201→
   202→  // Enhanced splash screen with progress bar animation
   203→  useEffect(() => {
   204→    // Animate progress bar
   205→    const progressInterval = setInterval(() => {
   206→      setSplashProgress((prev) => {
   207→        if (prev >= 100) {
   208→          clearInterval(progressInterval);
   209→          return 100;
   210→        }
   211→        // Non-linear progress: fast at start, slows down, then completes
   212→        const increment = prev < 30 ? 8 : prev < 60 ? 5 : prev < 85 ? 3 : 1;
   213→        return Math.min(prev + increment, 100);
   214→      });
   215→    }, 100);
   216→
   217→    // Initialize notifications during splash
   218→    initNotifications();
   219→
   220→    // Start splash exit after delay
   221→    const exitTimer = setTimeout(() => {
   222→      setSplashExiting(true);
   223→    }, 2000);
   224→
   225→    const doneTimer = setTimeout(() => {
   226→      setSplashDone(true);
   227→      // Check if onboarding should show
   228→      if (typeof window !== "undefined" && !localStorage.getItem("applenet_onboarding_done")) {
   229→        setShowOnboarding(true);
   230→      }
   231→    }, 2500);
   232→
   233→    return () => {
   234→      clearInterval(progressInterval);
   235→      clearTimeout(exitTimer);
   236→      clearTimeout(doneTimer);
   237→    };
   238→  }, []);
   239→
   240→  const handleOnboardingComplete = useCallback(() => {
   241→    setShowOnboarding(false);
   242→  }, []);
   243→
   244→  // Enhanced Splash Screen with progress bar, version, and shimmer
   245→  if (!splashDone) {
   246→    return (
   247→      <motion.div
   248→        initial={{ opacity: 1 }}
   249→        animate={{ opacity: splashExiting ? 0 : 1 }}
   250→        transition={{ duration: 0.5, ease: "easeInOut" }}
   251→        className="fixed inset-0 bg-white dark:bg-slate-900 flex flex-col items-center justify-center"
   252→        dir={isRTL ? "rtl" : "ltr"}
   253→      >
   254→        <motion.div
   255→          initial={{ scale: 0.5, opacity: 0 }}
   256→          animate={{ scale: 1, opacity: 1 }}
   257→          transition={{ type: "spring", stiffness: 200, damping: 15 }}
   258→          className="flex flex-col items-center"
   259→        >
   260→          {/* App Icon with shimmer */}
   261→          <div className="relative w-24 h-24 rounded-3xl overflow-hidden shadow-xl mb-6 border-2 border-[#E8F5E9] splash-shimmer">
   262→            <img src="/images/IMG_20260527_220851.jpg" alt="Apple.NET" className="w-full h-full object-cover" />
   263→          </div>
   264→          <AppleNetLogo size="lg" />
   265→
   266→          {/* Loading indicator */}
   267→          <motion.div
   268→            initial={{ opacity: 0 }}
   269→            animate={{ opacity: 1 }}
   270→            transition={{ delay: 0.5 }}
   271→            className="mt-6 flex items-center gap-2"
   272→          >
   273→            <div className="w-5 h-5 border-2 border-[#1B7A3D]/30 border-t-[#1B7A3D] rounded-full animate-spin" />
   274→            <span className="text-sm text-gray-400 dark:text-slate-500 font-bold">{t("splash.loading")}</span>
   275→          </motion.div>
   276→        </motion.div>
   277→
   278→        {/* Progress bar */}
   279→        <div className="absolute bottom-16 left-8 right-8 max-w-xs mx-auto">
   280→          <div className="h-1.5 bg-gray-100 rounded-full overflow-hidden">
   281→            <motion.div
   282→              initial={{ width: "0%" }}
   283→              animate={{ width: `${splashProgress}%` }}
   284→              transition={{ duration: 0.3, ease: "easeOut" }}
   285→              className="h-full bg-gradient-to-l from-[#1B7A3D] to-[#22A24D] rounded-full"
   286→            />
   287→          </div>
   288→        </div>
   289→
   290→        {/* Version display */}
   291→        <div className="absolute bottom-8 text-center">
   292→          <span className="text-[10px] text-gray-300 font-mono">v{APP_VERSION}</span>
   293→        </div>
   294→      </motion.div>
   295→    );
   296→  }
   297→
   298→  // Onboarding Flow
   299→  if (showOnboarding) {
   300→    return (
   301→      <OnboardingFlow
   302→        onComplete={handleOnboardingComplete}
   303→        onLoginClick={() => { setShowOnboarding(false); setAuthMode("login"); setShowAuth(true); }}
   304→        onRegisterClick={() => { setShowOnboarding(false); setAuthMode("register"); setShowAuth(true); }}
   305→      />
   306→    );
   307→  }
   308→
   309→  // PIN Lock Screen
   310→  if (pinLocked && user && pinEnabled) {
   311→    return (
   312→      <AnimatePresence>
   313→        <PinLockScreen
   314→          user={user}
   315→          onUnlock={() => setPinLocked(false)}
   316→          biometricEnabled={biometricEnabled}
   317→        />
   318→      </AnimatePresence>
   319→    );
   320→  }
   321→
   322→  return (
   323→    <div id="app-shell" className="font-sans bg-gray-50 dark:bg-slate-900" dir={isRTL ? "rtl" : "ltr"}>
   324→
   325→      {/* ===== App Update Banner ===== */}
   326→      <AppUpdateBanner />
   327→      <IOSInstallBanner />
   328→
   329→      {/* ===== STATUS BAR SIMULATION ===== */}
   330→      <div className="bg-white h-1 flex-shrink-0" />
   331→
   332→      {/* ===== LOCATION MODAL ===== */}
   333→      <AnimatePresence>
   334→        {showLocationModal && user && !locationModalDismissed && (
   335→          <motion.div
   336→            key="location-modal-overlay"
   337→            initial={{ opacity: 0 }}
   338→            animate={{ opacity: 1 }}
   339→            exit={{ opacity: 0 }}
   340→            className="fixed inset-0 z-[200] flex items-center justify-center bg-black/50 backdrop-blur-sm p-4"
   341→            dir="rtl"
   342→          >
   343→            <motion.div
   344→              key="location-modal-card"
   345→              initial={{ scale: 0.85, opacity: 0, y: 30 }}
   346→              animate={{ scale: 1, opacity: 1, y: 0 }}
   347→              exit={{ scale: 0.85, opacity: 0, y: 30 }}
   348→              transition={{ type: "spring", stiffness: 300, damping: 25 }}
   349→              className="w-full max-w-sm bg-white rounded-3xl overflow-hidden shadow-2xl"
   350→            >
   351→              {/* Green Header */}
   352→              <div className="bg-gradient-to-l from-[#1B7A3D] to-[#22A24D] p-5 text-center relative overflow-hidden">
   353→                <div className="absolute top-0 left-0 w-24 h-24 bg-white/5 rounded-full -translate-x-1/2 -translate-y-1/2" />
   354→                <div className="absolute bottom-0 right-0 w-16 h-16 bg-white/5 rounded-full translate-x-1/3 translate-y-1/3" />
   355→                <div className="relative z-10">
   356→                  <div className="w-14 h-14 rounded-2xl bg-white/20 flex items-center justify-center mx-auto mb-3">
   357→                    <MapPin className="w-7 h-7 text-white" />
   358→                  </div>
   359→                  <h2 className="text-xl font-black text-white mb-1">{t("location.title")}</h2>
   360→                  <p className="text-white/70 text-xs leading-relaxed">{t("location.subtitle")}</p>
   361→                </div>
   362→              </div>
   363→
   364→              {/* Form */}
   365→              <div className="p-5 space-y-4">
   366→                {/* Province */}
   367→                <div>
   368→                  <label className="block text-xs text-gray-500 mb-1.5 font-bold">{t("location.province")}</label>
   369→                  <div className="relative">
   370→                    <Globe className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   371→                    <select
   372→                      value={locationModalProvinceId}
   373→                      onChange={(e) => { setLocationModalProvinceId(e.target.value); setLocationModalDistrict(""); }}
   374→                      className="w-full bg-gray-50 border border-gray-200 text-gray-900 rounded-xl h-12 text-sm pr-10 pl-4 focus:border-[#1B7A3D] focus:ring-[#1B7A3D] appearance-none"
   375→                    >
   376→                      <option value="">{t("location.selectProvince")}</option>
   377→                      {PROVINCES.map(p => (
   378→                        <option key={p.id} value={p.id}>{lang === "en" && p.nameEn ? p.nameEn : p.name}</option>
   379→                      ))}
   380→                    </select>
   381→                  </div>
   382→                </div>
   383→
   384→                {/* District */}
   385→                <div>
   386→                  <label className="block text-xs text-gray-500 mb-1.5 font-bold">{t("location.district")}</label>
   387→                  <div className="relative">
   388→                    <Building2 className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   389→                    <select
   390→                      value={locationModalDistrict}
   391→                      onChange={(e) => setLocationModalDistrict(e.target.value)}
   392→                      disabled={!locationModalProvinceId}
   393→                      className="w-full bg-gray-50 border border-gray-200 text-gray-900 rounded-xl h-12 text-sm pr-10 pl-4 focus:border-[#1B7A3D] focus:ring-[#1B7A3D] appearance-none disabled:opacity-50 disabled:cursor-not-allowed"
   394→                    >
   395→                      <option value="">{locationModalProvinceId ? t("location.selectDistrict") : t("location.selectProvinceFirst")}</option>
   396→                      {locationModalProvinceId && (() => {
   397→                        const districtNames = lang === "en" ? getDistrictsEn(locationModalProvinceId) : getDistricts(locationModalProvinceId);
   398→                        const districtValues = getDistricts(locationModalProvinceId);
   399→                        return districtValues.map((d, i) => (
   400→                          <option key={d} value={d}>{districtNames[i]}</option>
   401→                        ));
   402→                      })()}
   403→                    </select>
   404→                  </div>
   405→                </div>
   406→
   407→                {/* Buttons */}
   408→                <div className="space-y-2 pt-1">
   409→                  <Button
   410→                    onClick={async () => {
   411→                      if (!locationModalProvinceId) {
   412→                        toast.error(t("location.selectProvinceFirst"));
   413→                        return;
   414→                      }
   415→                      setLocationModalSaving(true);
   416→                      try {
   417→                        const provinceObj = PROVINCES.find(p => p.id === locationModalProvinceId);
   418→                        await update(ref(db, `users/${user.uid}`), {
   419→                          provinceId: locationModalProvinceId,
   420→                          provinceName: provinceObj?.name || null,
   421→                          district: locationModalDistrict || null,
   422→                        });
   423→                        toast.success(t("location.savedSuccess"));
   424→                        setShowLocationModal(false);
   425→                        setLocationModalDismissed(true);
   426→                      } catch {
   427→                        toast.error(t("location.saveError"));
   428→                      }
   429→                      setLocationModalSaving(false);
   430→                    }}
   431→                    disabled={locationModalSaving}
   432→                    className="w-full bg-gradient-to-l from-[#1B7A3D] to-[#22A24D] hover:from-[#165E30] hover:to-[#134D28] text-white font-bold rounded-2xl h-12 btn-green-shadow disabled:opacity-70"
   433→                  >
   434→                    {locationModalSaving ? (
   435→                      <span className="flex items-center gap-2">
   436→                        <div className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" />
   437→                        {t("location.saving")}
   438→                      </span>
   439→                    ) : t("location.save")}
   440→                  </Button>
   441→                  <Button
   442→                    variant="ghost"
   443→                    onClick={() => {
   444→                      setShowLocationModal(false);
   445→                      setLocationModalDismissed(true);
   446→                    }}
   447→                    className="w-full text-gray-400 font-bold rounded-xl h-10 text-sm hover:text-gray-600"
   448→                  >
   449→                    {t("location.skip")}
   450→                  </Button>
   451→                </div>
   452→              </div>
   453→            </motion.div>
   454→          </motion.div>
   455→        )}
   456→      </AnimatePresence>
   457→
   458→      {/* ===== APP HEADER ===== */}
   459→      <header className="flex-shrink-0 app-header border-b border-gray-100/50 z-40">
   460→        <div className="max-w-lg mx-auto px-4 py-2.5 flex items-center justify-between">
   461→          {/* Menu Button */}
   462→          <motion.button
   463→            whileTap={{ scale: 0.85 }}
   464→            onClick={() => setShowMenu(!showMenu)}
   465→            className="w-10 h-10 rounded-2xl bg-[#E8F5E9] flex items-center justify-center hover:bg-green-100 transition-colors haptic-press"
   466→          >
   467→            <Menu className="w-5 h-5 text-[#1B7A3D]" />
   468→          </motion.button>
   469→
   470→          {/* App Logo */}
   471→          <AppleNetLogo size="sm" />
   472→
   473→          {/* Right Actions */}
   474→          <div className="flex items-center gap-1">
   475→            <ThemeToggle />
   476→            <LanguageToggle />
   477→            {user && <NotificationCenter uid={user.uid} isAdmin={isAdmin} />}
   478→            {user ? (
   479→              <motion.button
   480→                whileTap={{ scale: 0.85 }}
   481→                onClick={async () => { await signOut(auth); toast.success(t("auth.logout")); }}
   482→                className="w-10 h-10 rounded-2xl bg-red-50 dark:bg-red-900/20 flex items-center justify-center hover:bg-red-100 dark:hover:bg-red-900/30 transition-colors haptic-press"
   483→              >
   484→                <LogOut className="w-4 h-4 text-red-500" />
   485→              </motion.button>
   486→            ) : (
   487→              <motion.button
   488→                whileTap={{ scale: 0.85 }}
   489→                onClick={() => setShowAuth(true)}
   490→                className="w-10 h-10 rounded-2xl bg-[#E8F5E9] dark:bg-green-900/30 flex items-center justify-center hover:bg-green-100 dark:hover:bg-green-900/40 transition-colors haptic-press"
   491→              >
   492→                <LogIn className="w-4 h-4 text-[#1B7A3D]" />
   493→              </motion.button>
   494→            )}
   495→          </div>
   496→        </div>
   497→      </header>
   498→
   499→      {/* ===== SIDE MENU (Drawer) ===== */}
   500→      <AnimatePresence>
   501→        {showMenu && (
   502→          <>
   503→            <motion.div
   504→              initial={{ opacity: 0 }}
   505→              animate={{ opacity: 1 }}
   506→              exit={{ opacity: 0 }}
   507→              className="fixed inset-0 bg-black/40 z-[100]"
