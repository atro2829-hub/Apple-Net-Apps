    70→import { useLanguage } from "@/context/LanguageContext";
    71→
    72→const APP_VERSION = "2.1.0";
    73→
    74→export default function AppleNetApp() {
    75→  const { t, isRTL, lang } = useLanguage();
    76→
    77→  const NAV_TABS = [
    78→    { id: "home", icon: Home, label: t("nav.home") },
    79→    { id: "cards", icon: Wifi, label: t("nav.cards") },
    80→    { id: "starlink", icon: Satellite, label: "Starlink" },
    81→    { id: "chat", icon: MessageCircle, label: t("nav.chat") },
    82→    { id: "credit", icon: Wallet, label: t("nav.credit") },
    83→  ];
    84→
    85→  const [user, setUser] = useState<FirebaseUser | null>(null);
    86→  const [isAdmin, setIsAdmin] = useState(false);
    87→  const [userRole, setUserRole] = useState<string>("user");
    88→  const [managedNetwork, setManagedNetwork] = useState<string>("");
    89→  const [activeTab, setActiveTab] = useState("home");
    90→  const [showAuth, setShowAuth] = useState(false);
    91→  const [authMode, setAuthMode] = useState<"login" | "register">("login");
    92→  const [showAdmin, setShowAdmin] = useState(false);
    93→  const [loading, setLoading] = useState(true);
    94→  const [showMenu, setShowMenu] = useState(false);
    95→  const [splashDone, setSplashDone] = useState(false);
    96→  const [splashExiting, setSplashExiting] = useState(false);
    97→  const [userName, setUserName] = useState<string>("");
    98→  const [showOnboarding, setShowOnboarding] = useState(false);
    99→  const [splashProgress, setSplashProgress] = useState(0);
   100→  const [showLocationModal, setShowLocationModal] = useState(false);
   101→  const [locationModalProvinceId, setLocationModalProvinceId] = useState("");
   102→  const [locationModalDistrict, setLocationModalDistrict] = useState("");
   103→  const [locationModalSaving, setLocationModalSaving] = useState(false);
   104→  const [locationModalDismissed, setLocationModalDismissed] = useState(false);
   105→  const [userPhotoURL, setUserPhotoURL] = useState<string>("");
   106→  const [pinLocked, setPinLocked] = useState(false);
   107→  const [biometricEnabled, setBiometricEnabled] = useState(false);
   108→  const [pinEnabled, setPinEnabled] = useState(false);
   109→
   110→  // Listen for PIN/biometric setting changes
   111→  useEffect(() => {
   112→    if (!user) return;
   113→    const pinUnsub = onValue(ref(db, `users/${user.uid}/pinEnabled`), (snap) => {
   114→      setPinEnabled(!!snap.val());
   115→    });
   116→    const bioUnsub = onValue(ref(db, `users/${user.uid}/biometricEnabled`), (snap) => {
   117→      setBiometricEnabled(!!snap.val());
   118→    });
   119→    return () => { pinUnsub(); bioUnsub(); };
   120→  }, [user]);
   121→
   122→  // Handle URL params for PWA shortcuts
   123→  useEffect(() => {
   124→    if (typeof window === "undefined") return;
   125→    const params = new URLSearchParams(window.location.search);
   126→    const tab = params.get("tab");
   127→    if (tab && (NAV_TABS.some(t => t.id === tab) || tab === "profile" || tab === "deposit" || tab === "purchased" || tab === "submit-network")) {
   128→      setActiveTab(tab);
   129→    }
   130→  }, []);
   131→
   132→  useEffect(() => {
   133→    const unsub = onAuthStateChanged(auth, async (u) => {
   134→      setUser(u);
   135→      if (u) {
   136→        try {
   137→          const roleSnap = await get(ref(db, `users/${u.uid}/role`));
   138→          const role = roleSnap.val() || "user";
   139→          setUserRole(role);
   140→          setIsAdmin(role === "admin");
   141→          const nameSnap = await get(ref(db, `users/${u.uid}/displayName`));
   142→          setUserName(nameSnap.val() || u.displayName || u.email?.split("@")[0] || "م");
   143→          // Check if user has provinceId, show location modal if not
   144→          const provinceSnap = await get(ref(db, `users/${u.uid}/provinceId`));
   145→          const hasProvince = provinceSnap.exists() && provinceSnap.val();
   146→          if (!hasProvince && !locationModalDismissed) {
   147→            setShowLocationModal(true);
   148→          }
   149→          if (role === "network_manager") {
   150→            const netSnap = await get(ref(db, `users/${u.uid}/managedNetwork`));
   151→            setManagedNetwork(netSnap.val() || "");
   152→          }
   153→          // Check PIN and biometric settings
   154→          const pinSnap = await get(ref(db, `users/${u.uid}/pinEnabled`));
   155→          const bioSnap = await get(ref(db, `users/${u.uid}/biometricEnabled`));
   156→          const isPinEnabled = !!pinSnap.val();
   157→          const isBioEnabled = !!bioSnap.val();
   158→          setPinEnabled(isPinEnabled);
   159→          setBiometricEnabled(isBioEnabled);
   160→          if (isPinEnabled) setPinLocked(true);
   161→          // Initialize FCM
   162→          initFCMToken(u.uid).catch(() => {});
   163→        } catch {
   164→          setIsAdmin(false);
   165→          setUserRole("user");
   166→        }
   167→      } else {
   168→        setIsAdmin(false);
   169→        setUserRole("user");
   170→        setManagedNetwork("");
   171→        setUserName("");
   172→      }
   173→      setLoading(false);
   174→    });
   175→    return () => unsub();
   176→  }, []);
   177→
   178→  useEffect(() => {
   179→    if (!user) return;
   180→    const roleUnsub = onValue(ref(db, `users/${user.uid}/role`), (snap) => {
   181→      const role = snap.val() || "user";
   182→      setUserRole(role);
   183→      setIsAdmin(role === "admin");
   184→      if (role === "network_manager") {
   185→        get(ref(db, `users/${user.uid}/managedNetwork`)).then((netSnap) => {
   186→          setManagedNetwork(netSnap.val() || "");
   187→        });
   188→      } else {
   189→        setManagedNetwork("");
   190→      }
   191→    });
   192→    const nameUnsub = onValue(ref(db, `users/${user.uid}/displayName`), (snap) => {
   193→      if (snap.val()) setUserName(snap.val());
   194→    });
   195→    return () => { roleUnsub(); nameUnsub(); };
   196→  }, [user]);
   197→
   198→  // Listen for user photo URL
   199→  useEffect(() => {
   200→    if (!user) return;
   201→    const photoUnsub = onValue(ref(db, `users/${user.uid}/photoURL`), (snap) => {
   202→      if (snap.val()) setUserPhotoURL(snap.val());
   203→      else setUserPhotoURL("");
   204→    });
   205→    return () => photoUnsub();
   206→  }, [user]);
   207→
   208→  // Enhanced splash screen with progress bar animation
   209→  useEffect(() => {
   210→    // Animate progress bar
   211→    const progressInterval = setInterval(() => {
   212→      setSplashProgress((prev) => {
   213→        if (prev >= 100) {
   214→          clearInterval(progressInterval);
   215→          return 100;
   216→        }
   217→        // Non-linear progress: fast at start, slows down, then completes
   218→        const increment = prev < 30 ? 8 : prev < 60 ? 5 : prev < 85 ? 3 : 1;
   219→        return Math.min(prev + increment, 100);
   220→      });
   221→    }, 100);
   222→
   223→    // Initialize notifications during splash
   224→    initNotifications();
   225→
   226→    // Start splash exit after delay
   227→    const exitTimer = setTimeout(() => {
   228→      setSplashExiting(true);
   229→    }, 2000);
   230→
   231→    const doneTimer = setTimeout(() => {
   232→      setSplashDone(true);
   233→      // Check if onboarding should show
   234→      if (typeof window !== "undefined" && !localStorage.getItem("applenet_onboarding_done")) {
   235→        setShowOnboarding(true);
   236→      }
   237→    }, 2500);
   238→
   239→    return () => {
   240→      clearInterval(progressInterval);
   241→      clearTimeout(exitTimer);
   242→      clearTimeout(doneTimer);
   243→    };
   244→  }, []);
   245→
   246→  const handleOnboardingComplete = useCallback(() => {
   247→    setShowOnboarding(false);
   248→  }, []);
   249→
   250→  // Enhanced Splash Screen with progress bar, version, and shimmer
   251→  if (!splashDone) {
   252→    return (
   253→      <motion.div
   254→        initial={{ opacity: 1 }}
   255→        animate={{ opacity: splashExiting ? 0 : 1 }}
   256→        transition={{ duration: 0.5, ease: "easeInOut" }}
   257→        className="fixed inset-0 bg-white dark:bg-slate-900 flex flex-col items-center justify-center"
   258→        dir={isRTL ? "rtl" : "ltr"}
   259→      >
   260→        <motion.div
   261→          initial={{ scale: 0.5, opacity: 0 }}
   262→          animate={{ scale: 1, opacity: 1 }}
   263→          transition={{ type: "spring", stiffness: 200, damping: 15 }}
   264→          className="flex flex-col items-center"
   265→        >
   266→          {/* App Icon with shimmer */}
   267→          <div className="relative w-24 h-24 rounded-3xl overflow-hidden shadow-xl mb-6 border-2 border-[#E8F5E9] splash-shimmer">
   268→            <img src="/images/IMG_20260527_220851.jpg" alt="Apple.NET" className="w-full h-full object-cover" />
   269→          </div>
   270→          <AppleNetLogo size="lg" />
   271→
   272→          {/* Loading indicator */}
   273→          <motion.div
   274→            initial={{ opacity: 0 }}
   275→            animate={{ opacity: 1 }}
   276→            transition={{ delay: 0.5 }}
   277→            className="mt-6 flex items-center gap-2"
   278→          >
   279→            <div className="w-5 h-5 border-2 border-[#1B7A3D]/30 border-t-[#1B7A3D] rounded-full animate-spin" />
   280→            <span className="text-sm text-gray-400 dark:text-slate-500 font-bold">{t("splash.loading")}</span>
   281→          </motion.div>
   282→        </motion.div>
   283→
   284→        {/* Progress bar */}
   285→        <div className="absolute bottom-16 left-8 right-8 max-w-xs mx-auto">
   286→          <div className="h-1.5 bg-gray-100 rounded-full overflow-hidden">
   287→            <motion.div
   288→              initial={{ width: "0%" }}
   289→              animate={{ width: `${splashProgress}%` }}
   290→              transition={{ duration: 0.3, ease: "easeOut" }}
   291→              className="h-full bg-gradient-to-l from-[#1B7A3D] to-[#22A24D] rounded-full"
   292→            />
   293→          </div>
   294→        </div>
   295→
   296→        {/* Version display */}
   297→        <div className="absolute bottom-8 text-center">
   298→          <span className="text-[10px] text-gray-300 font-mono">v{APP_VERSION}</span>
   299→        </div>
   300→      </motion.div>
   301→    );
   302→  }
   303→
   304→  // Onboarding Flow
   305→  if (showOnboarding) {
   306→    return (
   307→      <OnboardingFlow
   308→        onComplete={handleOnboardingComplete}
   309→        onLoginClick={() => { setShowOnboarding(false); setAuthMode("login"); setShowAuth(true); }}
   310→        onRegisterClick={() => { setShowOnboarding(false); setAuthMode("register"); setShowAuth(true); }}
   311→      />
   312→    );
   313→  }
   314→
   315→  // PIN Lock Screen
   316→  if (pinLocked && user && pinEnabled) {
   317→    return (
   318→      <AnimatePresence>
   319→        <PinLockScreen
   320→          user={user}
   321→          onUnlock={() => setPinLocked(false)}
   322→          biometricEnabled={biometricEnabled}
   323→        />
   324→      </AnimatePresence>
   325→    );
   326→  }
   327→
   328→  return (
   329→    <div id="app-shell" className="font-sans bg-gray-50 dark:bg-slate-900" dir={isRTL ? "rtl" : "ltr"}>
   330→
   331→      {/* ===== App Update Banner ===== */}
   332→      <AppUpdateBanner />
   333→      <IOSInstallBanner />
   334→
   335→      {/* ===== STATUS BAR SIMULATION ===== */}
   336→      <div className="bg-white h-1 flex-shrink-0" />
   337→
   338→      {/* ===== LOCATION MODAL ===== */}
   339→      <AnimatePresence>
   340→        {showLocationModal && user && !locationModalDismissed && (
   341→          <motion.div
   342→            key="location-modal-overlay"
   343→            initial={{ opacity: 0 }}
   344→            animate={{ opacity: 1 }}
   345→            exit={{ opacity: 0 }}
   346→            className="fixed inset-0 z-[200] flex items-center justify-center bg-black/50 backdrop-blur-sm p-4"
   347→            dir="rtl"
   348→          >
   349→            <motion.div
   350→              key="location-modal-card"
   351→              initial={{ scale: 0.85, opacity: 0, y: 30 }}
   352→              animate={{ scale: 1, opacity: 1, y: 0 }}
   353→              exit={{ scale: 0.85, opacity: 0, y: 30 }}
   354→              transition={{ type: "spring", stiffness: 300, damping: 25 }}
   355→              className="w-full max-w-sm bg-white rounded-3xl overflow-hidden shadow-2xl"
   356→            >
   357→              {/* Green Header */}
   358→              <div className="bg-gradient-to-l from-[#1B7A3D] to-[#22A24D] p-5 text-center relative overflow-hidden">
   359→                <div className="absolute top-0 left-0 w-24 h-24 bg-white/5 rounded-full -translate-x-1/2 -translate-y-1/2" />
   360→                <div className="absolute bottom-0 right-0 w-16 h-16 bg-white/5 rounded-full translate-x-1/3 translate-y-1/3" />
   361→                <div className="relative z-10">
   362→                  <div className="w-14 h-14 rounded-2xl bg-white/20 flex items-center justify-center mx-auto mb-3">
   363→                    <MapPin className="w-7 h-7 text-white" />
   364→                  </div>
   365→                  <h2 className="text-xl font-black text-white mb-1">{t("location.title")}</h2>
   366→                  <p className="text-white/70 text-xs leading-relaxed">{t("location.subtitle")}</p>
   367→                </div>
   368→              </div>
   369→
   370→              {/* Form */}
   371→              <div className="p-5 space-y-4">
   372→                {/* Province */}
   373→                <div>
   374→                  <label className="block text-xs text-gray-500 mb-1.5 font-bold">{t("location.province")}</label>
   375→                  <div className="relative">
   376→                    <Globe className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   377→                    <select
   378→                      value={locationModalProvinceId}
   379→                      onChange={(e) => { setLocationModalProvinceId(e.target.value); setLocationModalDistrict(""); }}
   380→                      className="w-full bg-gray-50 border border-gray-200 text-gray-900 rounded-xl h-12 text-sm pr-10 pl-4 focus:border-[#1B7A3D] focus:ring-[#1B7A3D] appearance-none"
   381→                    >
   382→                      <option value="">{t("location.selectProvince")}</option>
   383→                      {PROVINCES.map(p => (
   384→                        <option key={p.id} value={p.id}>{lang === "en" && p.nameEn ? p.nameEn : p.name}</option>
   385→                      ))}
   386→                    </select>
   387→                  </div>
   388→                </div>
   389→
   390→                {/* District */}
   391→                <div>
   392→                  <label className="block text-xs text-gray-500 mb-1.5 font-bold">{t("location.district")}</label>
   393→                  <div className="relative">
   394→                    <Building2 className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   395→                    <select
   396→                      value={locationModalDistrict}
   397→                      onChange={(e) => setLocationModalDistrict(e.target.value)}
   398→                      disabled={!locationModalProvinceId}
   399→                      className="w-full bg-gray-50 border border-gray-200 text-gray-900 rounded-xl h-12 text-sm pr-10 pl-4 focus:border-[#1B7A3D] focus:ring-[#1B7A3D] appearance-none disabled:opacity-50 disabled:cursor-not-allowed"
   400→                    >
   401→                      <option value="">{locationModalProvinceId ? t("location.selectDistrict") : t("location.selectProvinceFirst")}</option>
   402→                      {locationModalProvinceId && (() => {
   403→                        const districtNames = lang === "en" ? getDistrictsEn(locationModalProvinceId) : getDistricts(locationModalProvinceId);
   404→                        const districtValues = getDistricts(locationModalProvinceId);
   405→                        return districtValues.map((d, i) => (
   406→                          <option key={d} value={d}>{districtNames[i]}</option>
   407→                        ));
   408→                      })()}
   409→                    </select>
   410→                  </div>
   411→                </div>
   412→
   413→                {/* Buttons */}
   414→                <div className="space-y-2 pt-1">
   415→                  <Button
   416→                    onClick={async () => {
   417→                      if (!locationModalProvinceId) {
   418→                        toast.error(t("location.selectProvinceFirst"));
   419→                        return;
   420→                      }
   421→                      setLocationModalSaving(true);
   422→                      try {
   423→                        const provinceObj = PROVINCES.find(p => p.id === locationModalProvinceId);
   424→                        await update(ref(db, `users/${user.uid}`), {
   425→                          provinceId: locationModalProvinceId,
   426→                          provinceName: provinceObj?.name || null,
   427→                          district: locationModalDistrict || null,
   428→                        });
   429→                        toast.success(t("location.savedSuccess"));
   430→                        setShowLocationModal(false);
   431→                        setLocationModalDismissed(true);
   432→                      } catch {
   433→                        toast.error(t("location.saveError"));
   434→                      }
   435→                      setLocationModalSaving(false);
   436→                    }}
   437→                    disabled={locationModalSaving}
   438→                    className="w-full bg-gradient-to-l from-[#1B7A3D] to-[#22A24D] hover:from-[#165E30] hover:to-[#134D28] text-white font-bold rounded-2xl h-12 btn-green-shadow disabled:opacity-70"
   439→                  >
   440→                    {locationModalSaving ? (
   441→                      <span className="flex items-center gap-2">
   442→                        <div className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" />
   443→                        {t("location.saving")}
   444→                      </span>
   445→                    ) : t("location.save")}
   446→                  </Button>
   447→                  <Button
   448→                    variant="ghost"
   449→                    onClick={() => {
   450→                      setShowLocationModal(false);
   451→                      setLocationModalDismissed(true);
   452→                    }}
   453→                    className="w-full text-gray-400 font-bold rounded-xl h-10 text-sm hover:text-gray-600"
   454→                  >
   455→                    {t("location.skip")}
   456→                  </Button>
   457→                </div>
   458→              </div>
   459→            </motion.div>
   460→          </motion.div>
   461→        )}
   462→      </AnimatePresence>
   463→
   464→      {/* ===== APP HEADER ===== */}
   465→      <header className="flex-shrink-0 app-header border-b border-gray-100/50 z-40">
   466→        <div className="max-w-lg mx-auto px-4 py-2.5 flex items-center justify-between">
   467→          {/* Menu Button */}
   468→          <motion.button
   469→            whileTap={{ scale: 0.85 }}
   470→            onClick={() => setShowMenu(!showMenu)}
   471→            className="w-10 h-10 rounded-2xl bg-[#E8F5E9] flex items-center justify-center hover:bg-green-100 transition-colors haptic-press"
   472→          >
   473→            <Menu className="w-5 h-5 text-[#1B7A3D]" />
   474→          </motion.button>
   475→
   476→          {/* App Logo */}
   477→          <AppleNetLogo size="sm" />
   478→
   479→          {/* Right Actions */}
   480→          <div className="flex items-center gap-1">
   481→            <ThemeToggle />
   482→            <LanguageToggle />
   483→            {user && <NotificationCenter uid={user.uid} isAdmin={isAdmin} />}
   484→            {user ? (
   485→              <motion.button
   486→                whileTap={{ scale: 0.85 }}
   487→                onClick={async () => { await signOut(auth); toast.success(t("auth.logout")); }}
   488→                className="w-10 h-10 rounded-2xl bg-red-50 dark:bg-red-900/20 flex items-center justify-center hover:bg-red-100 dark:hover:bg-red-900/30 transition-colors haptic-press"
   489→              >
   490→                <LogOut className="w-4 h-4 text-red-500" />
   491→              </motion.button>
   492→            ) : (
   493→              <motion.button
   494→                whileTap={{ scale: 0.85 }}
   495→                onClick={() => setShowAuth(true)}
   496→                className="w-10 h-10 rounded-2xl bg-[#E8F5E9] dark:bg-green-900/30 flex items-center justify-center hover:bg-green-100 dark:hover:bg-green-900/40 transition-colors haptic-press"
   497→              >
   498→                <LogIn className="w-4 h-4 text-[#1B7A3D]" />
   499→              </motion.button>
   500→            )}
   501→          </div>
   502→        </div>
   503→      </header>
